-- =====================================================================
-- STEP 1: CLEANUP / DROP EXISTING TABLES (Optional Reset)
-- =====================================================================
DROP TABLE IF EXISTS bookings CASCADE;
DROP TABLE IF EXISTS resorts CASCADE;
DROP TABLE IF EXISTS users CASCADE;
DROP TABLE IF EXISTS activities CASCADE;


-- =====================================================================
-- STEP 2: CREATE TABLES (Fully Matched with Frontend Schema)
-- =====================================================================

-- 1. USERS TABLE (Stores user profile details)
CREATE TABLE users (
  id TEXT PRIMARY KEY,
  name TEXT,
  email TEXT,
  phone TEXT,
  address TEXT,
  avatar TEXT,
  member_since TEXT,
  membership_tier TEXT DEFAULT 'Standard Member'
);

-- 2. RESORTS TABLE (Stores luxury resort listings)
CREATE TABLE resorts (
  id TEXT PRIMARY KEY,
  name TEXT,
  location TEXT,
  price NUMERIC,
  rating NUMERIC DEFAULT 4.5,
  reviews_count INTEGER DEFAULT 0,
  image TEXT,
  description TEXT,
  featured BOOLEAN DEFAULT true
);

-- 3. BOOKINGS TABLE (Stores user resort reservations)
CREATE TABLE bookings (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  resort_id TEXT,
  check_in DATE,
  check_out DATE,
  guests INTEGER DEFAULT 1,
  total_price NUMERIC,
  status TEXT DEFAULT 'Confirmed'
);

-- 4. ACTIVITIES TABLE (Stores resort events and account activity logs)
CREATE TABLE activities (
  id TEXT PRIMARY KEY,
  title TEXT,
  description TEXT,
  time TEXT
);


-- =====================================================================
-- STEP 3: DISABLE ROW LEVEL SECURITY (RLS) & ALLOW ALL PERMISSIONS
-- This prevents "new row violates row-level security policy" errors!
-- =====================================================================
ALTER TABLE users DISABLE ROW LEVEL SECURITY;
ALTER TABLE resorts DISABLE ROW LEVEL SECURITY;
ALTER TABLE bookings DISABLE ROW LEVEL SECURITY;
ALTER TABLE activities DISABLE ROW LEVEL SECURITY;

-- Fallback permissive security policies (in case RLS is re-enabled by Supabase)
DROP POLICY IF EXISTS "Allow public access on users" ON users;
DROP POLICY IF EXISTS "Allow public access on resorts" ON resorts;
DROP POLICY IF EXISTS "Allow public access on bookings" ON bookings;
DROP POLICY IF EXISTS "Allow public access on activities" ON activities;

CREATE POLICY "Allow public access on users" ON users FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access on resorts" ON resorts FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access on bookings" ON bookings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public access on activities" ON activities FOR ALL USING (true) WITH CHECK (true);
