-- ==========================================
-- 1. DROP EXISTING TABLES (Optional)
-- Run these if you need to reset your database
-- ==========================================
DROP TABLE IF EXISTS bookings;
DROP TABLE IF EXISTS resorts;
DROP TABLE IF EXISTS users;
DROP TABLE IF EXISTS activities;

-- ==========================================
-- 2. CREATE TABLES (Unrestricted for Workshop)
-- ==========================================

-- The Users Table
CREATE TABLE users (
  id TEXT PRIMARY KEY,
  name TEXT,
  email TEXT,
  phone TEXT,
  address TEXT,
  avatar TEXT,
  member_since TEXT,
  membership_tier TEXT
);

-- The Resorts Table
CREATE TABLE resorts (
  id TEXT PRIMARY KEY,
  name TEXT,
  location TEXT,
  price NUMERIC,
  rating NUMERIC,
  reviews_count INTEGER,
  image TEXT,
  description TEXT,
  featured BOOLEAN
);

-- The Bookings Table
-- NOTE: Foreign key constraints (REFERENCES) have been removed 
-- to prevent strict relationship errors during the student workshop.
CREATE TABLE bookings (
  id TEXT PRIMARY KEY,
  user_id TEXT,
  resort_id TEXT,
  check_in DATE,
  check_out DATE,
  guests INTEGER,
  total_price NUMERIC,
  status TEXT
);

-- The Activities Table
CREATE TABLE activities (
  id TEXT PRIMARY KEY,
  title TEXT,
  description TEXT,
  time TEXT
);

-- ==========================================
-- 2.5 DISABLE ROW LEVEL SECURITY
-- This ensures students don't get permission errors
-- ==========================================
ALTER TABLE users DISABLE ROW LEVEL SECURITY;
ALTER TABLE resorts DISABLE ROW LEVEL SECURITY;
ALTER TABLE bookings DISABLE ROW LEVEL SECURITY;
ALTER TABLE activities DISABLE ROW LEVEL SECURITY;
